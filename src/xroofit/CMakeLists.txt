# xRooFit checks ROOT_VERSION_CODE to find out which RooFit interfaces exist,
# e.g. if RooAbsArg::constOptimizeTestStatistic() is still there. In ROOT,
# that is the version of the RooFit it is built with. Here, it is the version of
# the RooFit-less ROOT installation, which can be older than the RooFit of this
# repository. Make the checks see the version of this RooFit instead, by
# force-including a header that redefines ROOT_VERSION_CODE. The override only
# affects the compilation of xRooFit itself, and none of its public headers
# check the version.
if(ROOT_VERSION VERSION_LESS PROJECT_VERSION)
  # Without leading zeros, which would make octal literals
  math(EXPR xroofit_version_minor "${PROJECT_VERSION_MINOR}")
  math(EXPR xroofit_version_patch "${PROJECT_VERSION_PATCH}")
  set(xroofit_version_header ${CMAKE_CURRENT_BINARY_DIR}/xRooFitRooFitVersion.h)
  file(CONFIGURE OUTPUT ${xroofit_version_header} CONTENT
"// Generated by src/xroofit/CMakeLists.txt, see there.
#include <ROOT/RVersion.hxx>
#undef ROOT_VERSION_CODE
#define ROOT_VERSION_CODE ROOT_VERSION(${PROJECT_VERSION_MAJOR}, ${xroofit_version_minor}, ${xroofit_version_patch})
")
  target_compile_options(RooFitXRooFit PRIVATE -include ${xroofit_version_header})
endif()
